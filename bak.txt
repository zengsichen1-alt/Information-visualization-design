const container2 = document.querySelector('.world-map-container');
const canvasMap = document.querySelector('.world-map');
const sceneMap = new THREE.Scene();
const cameraMap = new THREE.PerspectiveCamera(45,container2.clientWidth/container2.clientHeight,0.1,1000);
const rendererMap = new THREE.WebGLRenderer( { antialias : true, canvas: canvasMap } );
    let widthMap = container2.clientWidth;
    let heightMap = container2.clientHeight;
    cameraMap.aspect = widthMap / heightMap;//设置相机宽高比
    cameraMap.fov = 90;//设置相机视野
    cameraMap.updateProjectionMatrix();//更新相机投影矩阵
    rendererMap.setSize( widthMap, heightMap );//设置渲染尺寸
    rendererMap.setPixelRatio( Math.min(window.devicePixelRatio, 2) );//设置像素比

    window.addEventListener('resize', function() {
        let widthMap = container2.clientWidth;
        let heightMap = container2.clientHeight;
        cameraMap.aspect = widthMap / heightMap;
        cameraMap.updateProjectionMatrix();
        rendererMap.setSize( widthMap, heightMap );
        rendererMap.setPixelRatio( Math.min(window.devicePixelRatio, 2) );//设置像素比
    });
    
    cameraMap.position.set(0, 0, 60);
    cameraMap.lookAt(0, 100, 0);
    sceneMap.add(cameraMap);
    const loder = new GLTFLoader();
    loder.load('model/earthLow.glb', function(gltf){
            gltf.scene.position.set(0,0,0);
            gltf.scene.rotation.set(0,0,0);
            sceneMap.add(gltf.scene);
    });

   

    const controlsMap = new OrbitControls(cameraMap, rendererMap.domElement);
    
    
    controlsMap.enableDamping = true;
    controlsMap.dampingFactor = 0.05; // 阻尼系数
    controlsMap.enablePan = false;    // 禁止右键平移
    controlsMap.enableZoom = false;    // 是否允许滚轮缩放
    controlsMap.minDistance = 30;     // 最小缩放距离
    controlsMap.maxDistance = 150;    // 最大缩放距离
    controlsMap.target.set(0, 0, 0);

    //光照
    const areaLight = new THREE.RectAreaLight(0xffffff, 0.0, 20, 20);
    areaLight.position.set(0, 0, -1);
    areaLight.lookAt(0, 0, 0);
    cameraMap.add(areaLight);

    const pointLight = new THREE.PointLight(0xffFFFF, 50, 0, 2);
    pointLight.position.set(-20, -30, 0);
    sceneMap.add(pointLight);

    rendererMap.outputColorSpace = THREE.SRGBColorSpace;//颜色空间
    rendererMap.toneMapping = THREE.ACESFilmicToneMapping;//色调映射
    rendererMap.toneMappingExposure = 2.0;//曝光度
    rendererMap.shadowMap.enabled = true;//开启阴影
    rendererMap.shadowMap.type = THREE.PCFSoftShadowMap;//阴影类型
    const sunLightMap = new THREE.DirectionalLight(0xFFFFFF, 3);
    sunLightMap.position.set(0, 0, -1);
    sunLightMap.target.position.set(0, 0, 0);
    sunLightMap.castShadow = true;
    sunLightMap.shadow.camera.near = 0.5;//近端
    sunLightMap.shadow.mapSize.width = 1024;//阴影质量
    sunLightMap.shadow.mapSize.height = 1024;//阴影质量
    sunLightMap.shadow.bias = -0.0001;
    cameraMap.add(sunLightMap);
    const ambient = new THREE.AmbientLight(0xffffff,2);
    sceneMap.add(ambient);
    //光照

    

    function mapanimate(){
            requestAnimationFrame( mapanimate );
            controlsMap.update();
            
            rendererMap.render( sceneMap, cameraMap );
    }
    mapanimate();
